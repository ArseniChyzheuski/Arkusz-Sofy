FROM meble

Zapytanie 3
SELECT meble.nazwa, meble.cena
FROM meble
JOIN zakupy ON meble.idMeble = zakupy.idMeble
WHERE zakupy.idKlienci = 1;
Zapytanie 4 
SELECT meble.nazwa, klienci.imie, klienci.nazwisko
FROM meble
JOIN zakupy ON meble.idMeble = zakupy.idMeble
JOIN klienci ON zakupy.idKlienci = klienci.idKlienci;
